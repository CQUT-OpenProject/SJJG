fun MergeList(La, Lb):
    pa = La.next
    pb = Lb.next
    pc = La

    while pa != NULL and pb != NULL:
        if pa.data <= pb.data:
            pc.next = pa
            pc = pa
            pa = pa.next
        else:
            pc.next = pb
            pc = pb
            pb = pb.next

    if pa != NULL:
        pc.next = pa
    else:
        pc.next = pb

    return La
